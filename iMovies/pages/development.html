⚙️ Controller Examples

controllers/movieController.js 
const Movie = require('../models/Movie');

// GET /api/movies — with genre, language, date filters
exports.getMovies = async (req, res) => {
  try {
    const { genre, language, search } = req.query;
    const query = { isActive: true };

    if (genre)    query.genre    = genre;
    if (language) query.language = language;
    if (search)   query.$text    = { $search: search };

    const movies = await Movie.find(query)
      .sort({ releaseDate: -1 })
      .limit(20);

    res.json({ success: true, count: movies.length, data: movies });
  } catch (err) {
    res.status(500).json({ success: false, message: err.message });
  }
};

// POST /api/movies — admin only
exports.createMovie = async (req, res) => {
  try {
    const movie = await Movie.create(req.body);
    res.status(201).json({ success: true, data: movie });
  } catch (err) {
    res.status(400).json({ success: false, message: err.message });
  }
};
controllers/bookingController.js 
const Booking = require('../models/Booking');
const Show    = require('../models/Show');

// POST /api/bookings — create booking
exports.createBooking = async (req, res) => {
  try {
    const { showId, seats } = req.body;

    // Check show exists and seats are available
    const show = await Show.findById(showId);
    if (!show) return res.status(404).json({ message: 'Show not found' });

    const alreadyBooked = show.bookedSeats.some(s => seats.includes(s));
    if (alreadyBooked) return res.status(400).json({ message: 'Some seats already booked' });

    const totalAmount = seats.length * show.ticketPrice;

    const booking = await Booking.create({
      user: req.user._id,
      show: showId,
      seats,
      totalAmount,
      status: 'confirmed',
    });

    // Mark seats as booked in show
    show.bookedSeats.push(...seats);
    await show.save();

    res.status(201).json({ success: true, data: booking });
  } catch (err) {
    res.status(500).json({ success: false, message: err.message });
  }
};

🛣️ Routes Example

routes/movieRoutes.js 
const express = require('express');
const router  = express.Router();
const { protect, adminOnly } = require('../middleware/authMiddleware');
const {
  getMovies, getMovie, createMovie, updateMovie, deleteMovie
} = require('../controllers/movieController');

router.route('/')
  .get(getMovies)
  .post(protect, adminOnly, createMovie);

router.route('/:id')
  .get(getMovie)
  .put(protect, adminOnly, updateMovie)
  .delete(protect, adminOnly, deleteMovie);

module.exports = router;
