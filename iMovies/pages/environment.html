🚂 Backend Setup (Express.js)

Step 1 – Initialize the Backend 
mkdir imovies && cd imovies
mkdir backend && cd backend
npm init -y
Step 2 – Install Backend Dependencies 
npm install express mongoose bcryptjs jsonwebtoken cors dotenv multer socket.io

# Dev dependencies
npm install --save-dev nodemon
Step 3 – server.js (Entry Point) 
const express = require('express');
const mongoose = require('mongoose');
const cors = require('cors');
const http = require('http');
const { Server } = require('socket.io');
require('dotenv').config();

const app = express();
const server = http.createServer(app);
const io = new Server(server, { cors: { origin: '*' } });

app.use(cors());
app.use(express.json());

// Routes
app.use('/api/auth',    require('./routes/authRoutes'));
app.use('/api/movies',  require('./routes/movieRoutes'));
app.use('/api/booking', require('./routes/bookingRoutes'));

// MongoDB Connection
mongoose.connect(process.env.MONGO_URI)
  .then(() => console.log('MongoDB connected'))
  .catch(err => console.error(err));

// Socket.io – Real-time seat updates
io.on('connection', (socket) => {
  socket.on('seat-selected', (data) => {
    socket.broadcast.emit('seat-updated', data);
  });
});

const PORT = process.env.PORT || 5000;
server.listen(PORT, () => console.log(`Server running on port ${PORT}`));

⚛️ Frontend Setup (React.js)

Step 1 – Create React App 
cd ..  # back to imovies/
npx create-react-app frontend
cd frontend
Step 2 – Install Frontend Dependencies 
npm install axios react-router-dom redux react-redux react-toastify bootstrap socket.io-client
Step 3 – src/index.js 
import React from 'react';
import ReactDOM from 'react-dom/client';
import { Provider } from 'react-redux';
import store from './redux/store';
import App from './App';
import 'bootstrap/dist/css/bootstrap.min.css';
import 'react-toastify/dist/ReactToastify.css';
import './index.css';

const root = ReactDOM.createRoot(document.getElementById('root'));
root.render(
  <Provider store={store}>
    <App />
  </Provider>
);
